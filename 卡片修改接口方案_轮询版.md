# 卡片修改接口方案（轮询版）

## 一、总体架构

### 1.1 系统流程
```
前端选择元素 → 后端接收请求 → 调用Claude CLI → 修改HTML文件 → 监控文件变化 → 轮询查询状态
```

### 1.2 核心组件
- **前端**: HtmlEditModal组件（已实现）
- **后端API**: Express服务端点
- **AI服务**: 直接调用 Claude CLI (`claude --dangerously-skip-permissions -p`)
- **文件监控**: 文件系统监听器
- **状态管理**: 轮询机制

## 二、接口设计

## 三、后端实现

### 3.1 HtmlEditService类结构

```javascript
class HtmlEditService {
  constructor() {
    this.monitor = new FileChangeMonitor();
    this.claudeExecutor = new ClaudeDirectExecutor();
  }

  // 处理编辑请求
  async processEditRequest(request, userId) {
    const taskId = `task_${Date.now()}_${uuidv4().substring(0, 8)}`;

    try {
      // 1. 构建完整路径
      const userWorkspace = `/app/data/users/${userId || 'default'}/workspace/`;
      const fullPath = path.join(userWorkspace, request.htmlPath);

      // 2. 读取原始HTML
      const originalContent = await fs.readFile(fullPath, 'utf-8');

      // 3. 备份文件
      const backupPath = await this.backupFile(fullPath);

      // 4. 构建提示词
      const prompt = this.buildPrompt(fullPath, request.elements, request.request, originalContent);

      // 5. 调用Claude CLI
      const modifiedHtml = await this.claudeExecutor.execute(prompt);

      // 6. 写入修改后的文件
      await fs.writeFile(fullPath, modifiedHtml, 'utf-8');

      // 7. 开始监控文件变化
      this.monitor.watchFile(fullPath, taskId);

      return {
        success: true,
        taskId,
        status: 'processing',
        backupPath
      };
    } catch (error) {
      this.monitor.taskStatuses.set(taskId, {
        status: 'failed',
        error: error.message
      });
      throw error;
    }
  }

  // 获取任务状态
  getTaskStatus(taskId) {
    return this.monitor.getTaskStatus(taskId);
  }
}
```

### 3.2 Claude直接执行器

```javascript
class ClaudeDirectExecutor {
  async execute(prompt) {
    const { spawn } = require('child_process');

    return new Promise((resolve, reject) => {
      // 使用 -p 参数直接传递提示词
      const claude = spawn('claude', [
        '--dangerously-skip-permissions',
        '-p',
        prompt
      ], {
        env: {
          ...process.env,
          ANTHROPIC_AUTH_TOKEN: process.env.ANTHROPIC_AUTH_TOKEN
        },
        maxBuffer: 10 * 1024 * 1024 // 10MB buffer
      });

      let output = '';
      let errorOutput = '';

      claude.stdout.on('data', (data) => {
        output += data.toString();
      });

      claude.stderr.on('data', (data) => {
        errorOutput += data.toString();
      });

      claude.on('close', (code) => {
        if (code === 0 && output) {
          const htmlContent = this.extractHtmlFromOutput(output);
          if (this.validateHtml(htmlContent)) {
            resolve(htmlContent);
          } else {
            reject(new Error('Invalid HTML output'));
          }
        } else {
          reject(new Error(errorOutput || `Process exited with code ${code}`));
        }
      });

      claude.on('error', (error) => {
        reject(error);
      });
    });
  }

  extractHtmlFromOutput(output) {
    // 移除markdown标记
    let html = output.replace(/```html?\n?/g, '').replace(/```\n?/g, '');

    // 找到HTML开始
    const htmlStart = html.match(/<!DOCTYPE html|<html/i);
    if (htmlStart) {
      html = html.substring(html.indexOf(htmlStart[0]));
    }

    // 找到HTML结束
    const htmlEnd = html.lastIndexOf('</html>');
    if (htmlEnd !== -1) {
      html = html.substring(0, htmlEnd + 7);
    }

    return html.trim();
  }

  validateHtml(html) {
    if (!html || html.length < 100) return false;

    const hasHtmlTag = /<html[\s>]/i.test(html) || /<!DOCTYPE\s+html/i.test(html);
    const hasClosingTag = /<\/html>/i.test(html);

    return hasHtmlTag || hasClosingTag;
  }
}
```

### 3.3 文件监控器

```javascript
class FileChangeMonitor {
  constructor() {
    this.watchers = new Map();
    this.taskStatuses = new Map();
  }

  watchFile(filePath, taskId) {
    // 初始化任务状态
    this.taskStatuses.set(taskId, {
      status: 'processing',
      progress: 50,
      startedAt: new Date().toISOString()
    });

    // 监听文件变化
    const watcher = fs.watch(filePath, (eventType) => {
      if (eventType === 'change') {
        this.taskStatuses.set(taskId, {
          status: 'completed',
          progress: 100,
          completedAt: new Date().toISOString()
        });

        // 清理监听器
        watcher.close();
        this.watchers.delete(taskId);
      }
    });

    this.watchers.set(taskId, watcher);

    // 设置60秒超时
    setTimeout(() => {
      if (this.watchers.has(taskId)) {
        this.taskStatuses.set(taskId, {
          status: 'failed',
          error: 'Timeout waiting for file change'
        });
        watcher.close();
        this.watchers.delete(taskId);
      }
    }, 60000);
  }

  getTaskStatus(taskId) {
    return this.taskStatuses.get(taskId) || {
      status: 'unknown',
      error: 'Task not found'
    };
  }
}
```

### 3.4 路由处理

```javascript
// routes/htmlEdit.js
const express = require('express');
const router = express.Router();
const editService = new HtmlEditService();

// 提交编辑请求
router.post('/edit', async (req, res) => {
  try {
    const { htmlPath, fileId, folderId, elements, request } = req.body;

    if (!htmlPath || !elements || !request) {
      return res.status(400).json({
        success: false,
        error: 'Missing required parameters'
      });
    }

    const userId = req.user?.id || 'default';

    const result = await editService.processEditRequest(
      { htmlPath, fileId, folderId, elements, request },
      userId
    );

    res.json(result);
  } catch (error) {
    console.error('HTML edit error:', error);
    res.status(500).json({
      success: false,
      error: error.message
    });
  }
});

// 查询任务状态
router.get('/status/:taskId', (req, res) => {
  try {
    const { taskId } = req.params;
    const status = editService.getTaskStatus(taskId);

    res.json(status);
  } catch (error) {
    res.status(500).json({
      success: false,
      error: error.message
    });
  }
});

module.exports = router;
```

## 四、前端实现

### 4.1 提交修改请求

```javascript
// HtmlEditModal.vue
const handleApply = async () => {
  if (selectedElements.value.length === 0) {
    ElMessage.warning('请先选择要修改的元素');
    return;
  }

  if (!editRequest.value.trim()) {
    ElMessage.warning('请输入修改内容');
    return;
  }

  isApplying.value = true;

  try {
    // 转换为相对路径
    const username = localStorage.getItem('username') || 'default';
    let relativePath = props.htmlPath;
    const workspacePrefix = `/app/data/users/${username}/workspace/`;
    if (relativePath.startsWith(workspacePrefix)) {
      relativePath = relativePath.substring(workspacePrefix.length);
    }

    const requestData = {
      htmlPath: relativePath,
      fileId: props.fileId,
      folderId: props.folderId,
      elements: selectedElements.value.map(item => ({
        tagName: item.element.tagName,
        className: item.element.className,
        id: item.element.id,
        html: item.element.outerHTML,
        coverage: item.coverage
      })),
      request: editRequest.value,
      timestamp: Date.now()
    };

    const response = await fetch('/api/html/edit', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify(requestData)
    });

    const result = await response.json();

    if (result.success) {
      // 开始轮询任务状态
      pollTaskStatus(result.taskId);

      ElMessage.info({
        message: '修改任务已提交，正在处理中...',
        duration: 3000
      });
    } else {
      throw new Error(result.error || '提交失败');
    }
  } catch (error) {
    console.error('提交修改失败:', error);
    ElMessage.error('提交失败，请重试');
    isApplying.value = false;
  }
};
```

### 4.2 轮询任务状态

```javascript
const pollTaskStatus = async (taskId) => {
  let retryCount = 0;
  const maxRetries = 30; // 最多60秒
  const pollInterval = 2000; // 每2秒查询一次

  const checkStatus = async () => {
    try {
      const response = await fetch(`/api/html/status/${taskId}`);
      const data = await response.json();

      if (data.status === 'completed') {
        ElMessage.success('修改已完成');

        // 触发应用成功事件
        emit('apply', {
          status: 'completed',
          taskId,
          htmlPath: props.htmlPath
        });

        // 关闭对话框
        visible.value = false;
        isApplying.value = false;
        return;
      } else if (data.status === 'failed') {
        ElMessage.error(`修改失败: ${data.error || '未知错误'}`);
        isApplying.value = false;
        return;
      } else if (retryCount >= maxRetries) {
        ElMessage.warning('修改超时，请稍后查看结果');
        isApplying.value = false;
        return;
      }

      // 继续轮询
      retryCount++;
      setTimeout(checkStatus, pollInterval);

    } catch (error) {
      console.error('查询任务状态失败:', error);

      // 网络错误重试
      if (retryCount < 3) {
        retryCount++;
        setTimeout(checkStatus, pollInterval * 2);
      } else {
        ElMessage.error('网络错误，无法查询任务状态');
        isApplying.value = false;
      }
    }
  };

  // 开始轮询
  checkStatus();
};
```

## 十、总结

本方案通过直接调用Claude CLI，实现了独立、高效的HTML编辑功能：

1. **独立性**: 不依赖其他API服务
2. **可靠性**: 直接控制Claude调用和输出处理
3. **效率**: 减少中间层，提高响应速度
4. **灵活性**: 完全控制提示词格式和输出处理

轮询机制确保了前端能及时获取任务状态，文件监控确保了修改的可靠性。
